% Competing-analyses version of the evidence-to-policy diagram: one piece of
% research feeds three rival policy analyses, and each policy maker picks the
% one that suits them. Redrawn in the style of evidence-to-policy.tex from the
% dissertation figure (dissertation-manuscript/dissertation.tex, label
% med_cred). Built by ./build.sh as the "competing" variant. With
% \fromanalysis defined it starts at the policy analyses (no Truth, no
% Research, no middle analysis, no "Observed by citizens" brace): the
% "competing-from-analysis" variant.

\documentclass[border=10pt,tikz]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning,shapes.geometric,decorations.pathreplacing,calc}
\pdftrailerid{}

% The from-analysis variant calls the policy makers "decision makers".
\ifdefined\fromanalysis\def\maker{Decision}\else\def\maker{Policy}\fi

% \cubex: x of the two analysis cubes; the build sets -2 for the blackbox and opa
% variants (cubes 2cm further left, the rest in place) and leaves 0 elsewhere.
\ifdefined\cubex\else\def\cubex{0}\fi
% \rowtop, \rowbot: the y of the two rows (default +/-2.4); \midx: x of the third OPA.
\ifdefined\rowtop\else\def\rowtop{2.4}\fi
\ifdefined\rowbot\else\def\rowbot{-2.4}\fi
\ifdefined\midx\else\def\midx{1.6}\fi

\begin{document}
\begin{tikzpicture}[
  every node/.style={font=\large},
  bigcirc/.style={circle, draw, minimum size=2.4cm, align=center, inner sep=2pt},
  tri/.style={regular polygon, regular polygon sides=3, draw,
              minimum size=2.5cm, align=center, inner sep=0pt},
  rrect/.style={rectangle, rounded corners=8pt, draw, minimum width=2cm,
                minimum height=1cm, align=center},
  arr/.style={-stealth, thick},
  darr/.style={-stealth, thick, dashed},
  note/.style={font=\normalsize\itshape}
]

\ifdefined\fromanalysis
% Two rival analyses only, numbered 1 and 2 (no middle "gains & losses" one).
% At (0, 2.4) and (0, -2.4), so the policy makers land where the web-from-analysis
% variant puts them (4.8, 2.4) and (4.8, -2.4); the rows are far enough apart for the
% 3.6cm cubes and the third OPA between them.
% With \blackbox also defined, the two analyses are closed dark cubes of the OPA icon's
% shape: the "competing-from-analysis-blackbox" variant.
% With \opaicon defined, each analysis is the Open Policy Analysis icon (static/opa-all.png,
% the four-layer stack from opa-prop40), in a 2.4cm cell so the arrows start where they did:
% the "competing-from-analysis-opa" variant.
\tikzset{pa/.style={bigcirc}}
\ifdefined\blackbox
% A closed cube with the OPA icon's outline (static/opa-closed.png: the same 516 x 600 shape,
% three dark faces, no layers open), drawn 3.6cm tall like the OPA icon (50% above the
% 2.4cm circles).
\node[minimum size=2.4cm, inner sep=0pt] (PA1) at (\cubex,\rowtop)  {\includegraphics[height=3.6cm]{static/opa-closed.png}};
\node[minimum size=2.4cm, inner sep=0pt] (PA3) at (\cubex,\rowbot) {\includegraphics[height=3.6cm]{static/opa-closed.png}};
% The labels lie on each cube's top face: its centre is 0.91cm above the cube's centre, and
% its edges run along (1.27, +/-0.89), so the text axes are those edge directions.
\foreach \c/\n in {PA1/1, PA3/2}
  \node[text=white, font=\scriptsize\bfseries, align=center, inner sep=0pt,
        cm={0.819,0.574,-0.819,0.574,(0,0)}, scale=0.85] at ($(\c.center)+(0,0.91)$) {Policy\\Analysis \n};
\else\ifdefined\opaicon
\node[minimum size=2.4cm, inner sep=0pt] (PA1) at (\cubex,\rowtop)  {\includegraphics[height=3.6cm]{static/opa-all.png}};
\node[minimum size=2.4cm, inner sep=0pt] (PA3) at (\cubex,\rowbot) {\includegraphics[height=3.6cm]{static/opa-all.png}};
% With \thirdopa also defined, a third OPA of the same size sits between the two rows, with
% solid arrows in from both analyses and out to both decision makers (drawn below, after the
% makers):
% the "competing-from-analysis-opa-third" variant.
\ifdefined\thirdopa
\node[minimum size=2.4cm, inner sep=0pt] (PAM) at (\midx,{(\rowtop+\rowbot)/2}) {\includegraphics[height=3.6cm]{static/opa-all.png}};
\fi
\else
\node[pa]                                  (PA1) at (0,2.4)  {Policy\\Analysis 1};
\node[pa]                                  (PA3) at (0,-2.4) {Policy\\Analysis 2};
\fi\fi
\else
\node[tri]                                  (T)   {Truth};
\node[bigcirc, right=1.2cm of T]            (R)   {Research};
\node[bigcirc, right=1.8cm of R]            (PA2) {Policy\\Analysis 2};
\node[bigcirc, above=0.4cm of PA2]          (PA1) {Policy\\Analysis 1};
\node[bigcirc, below=0.4cm of PA2]          (PA3) {Policy\\Analysis 3};
\fi
% In the from-analysis variants the makers sit at fixed coordinates, so wider analysis
% nodes (the 3.6cm cubes) do not push them right.
\ifdefined\fromanalysis
\node[bigcirc]                              (PM1) at (4.8,\rowtop)  {\maker\\Maker 1};
\node[bigcirc]                              (PM2) at (4.8,\rowbot) {\maker\\Maker 2};
\else
\node[bigcirc, right=2.4cm of PA1]          (PM1) {\maker\\Maker 1};
\node[bigcirc, right=2.4cm of PA3]          (PM2) {\maker\\Maker 2};
\fi
\node[rrect,   right=1.5cm of PM1]          (S)   {Support};
\node[rrect,   right=1.5cm of PM2]          (O)   {Oppose};

\ifdefined\fromanalysis\else
\draw[arr]  (T)   -- (R);
\draw[darr] (R)   -- (PA1);
\draw[arr]  (R)   -- (PA2);
\draw[darr] (R)   -- (PA3);
\fi
% The third-OPA variant has no direct arrows from the two analyses to the makers.
\ifdefined\thirdopa\else
\draw[arr]  (PA1) -- node[note, above] {only gains} (PM1);
\draw[arr]  (PA3) -- node[note, above] {only losses} (PM2);
\fi
\ifdefined\fromanalysis\else
\node[note, right=0.3cm of PA2] {gains \& losses};
\fi
\draw[arr]  (PM1) -- (S);
\draw[arr]  (PM2) -- (O);

% What citizens see: the rival analyses and what each policy maker does.
% Left out of the competing-from-analysis variant.
\ifdefined\fromanalysis\else
\coordinate (bl) at ($(PA1.north west)+(0,0.5)$);
\draw[decorate, decoration={brace, amplitude=8pt}, thick]
  (bl) -- (bl -| S.east) node[midway, above=10pt] {Observed by citizens};
\fi

% One frame for every from-analysis variant (the 3.6cm cubes reach x = -1.55 and
% y = +/-4.2), so all of them have the same size and line up on slides.
\ifdefined\thirdopa
% Arrows meet the icons' visible corners (the icon's side corners sit 1.27cm left or right of
% its centre and 0.9cm above or below it), not the transparent edge of the image.
\draw[arr, shorten <=2pt, shorten >=2pt] ($(PA1.center)+(1.27,-0.9)$) -- ($(PAM.center)+(-1.27,0.9)$);
\draw[arr, shorten <=2pt, shorten >=2pt] ($(PA3.center)+(1.27,0.9)$)  -- ($(PAM.center)+(-1.27,-0.9)$);
\draw[arr, shorten <=2pt] ($(PAM.center)+(1.27,0.9)$)  -- (PM1);
\draw[arr, shorten <=2pt] ($(PAM.center)+(1.27,-0.9)$) -- (PM2);
\fi
\ifdefined\fromanalysis\path ({\cubex-1.56},{\rowtop+1.81}) ({\cubex-1.56},{\rowbot-1.81});\fi
\end{tikzpicture}
\end{document}
