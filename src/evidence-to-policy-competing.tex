% Competing-analyses version of the evidence-to-policy diagram: one piece of
% research feeds three rival policy analyses, and each policy maker picks the
% one that suits them. Redrawn in the style of evidence-to-policy.tex from the
% dissertation figure (dissertation-manuscript/dissertation.tex, label
% med_cred). Built by ./build.sh as the "competing" variant. With
% \fromanalysis defined it starts at the policy analyses (no Truth, no
% Research, no middle analysis, no "Observed by citizens" brace): the
% "competing-from-analysis" variant.

\documentclass[border=10pt,tikz]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning,shapes.geometric,decorations.pathreplacing,calc}
\pdftrailerid{}

% The from-analysis variant calls the policy makers "decision makers".
\ifdefined\fromanalysis\def\maker{Decision}\else\def\maker{Policy}\fi

\begin{document}
\begin{tikzpicture}[
  every node/.style={font=\large},
  bigcirc/.style={circle, draw, minimum size=2.4cm, align=center, inner sep=2pt},
  tri/.style={regular polygon, regular polygon sides=3, draw,
              minimum size=2.5cm, align=center, inner sep=0pt},
  rrect/.style={rectangle, rounded corners=8pt, draw, minimum width=2cm,
                minimum height=1cm, align=center},
  arr/.style={-stealth, thick},
  darr/.style={-stealth, thick, dashed},
  note/.style={font=\normalsize\itshape}
]

\ifdefined\fromanalysis
% Two rival analyses only, numbered 1 and 2 (no middle "gains & losses" one).
% At (0, 1.8) and (0, -1.8), so the policy makers land where the web-from-analysis
% variant puts them (4.8, 1.8) and (4.8, -1.8).
% With \blackbox also defined, the two analyses are drawn as opaque dark boxes of the
% same size: the "competing-from-analysis-blackbox" variant.
% With \opaicon defined, each analysis is the Open Policy Analysis icon (static/opa-all.png,
% the four-layer stack from opa-prop40), in a 2.4cm cell so the arrows start where they did:
% the "competing-from-analysis-opa" variant.
\ifdefined\blackbox
\tikzset{pa/.style={rectangle, rounded corners=3pt, draw=black, fill=black!82, text=white,
                    minimum size=2.4cm, align=center, inner sep=2pt}}
\else
\tikzset{pa/.style={bigcirc}}
\fi
\ifdefined\opaicon
\node[minimum size=2.4cm, inner sep=0pt] (PA1) at (0,1.8)  {\includegraphics[height=2.4cm]{static/opa-all.png}};
\node[minimum size=2.4cm, inner sep=0pt] (PA3) at (0,-1.8) {\includegraphics[height=2.4cm]{static/opa-all.png}};
\else
\node[pa]                                  (PA1) at (0,1.8)  {Policy\\Analysis 1};
\node[pa]                                  (PA3) at (0,-1.8) {Policy\\Analysis 2};
\fi
\else
\node[tri]                                  (T)   {Truth};
\node[bigcirc, right=1.2cm of T]            (R)   {Research};
\node[bigcirc, right=1.8cm of R]            (PA2) {Policy\\Analysis 2};
\node[bigcirc, above=0.4cm of PA2]          (PA1) {Policy\\Analysis 1};
\node[bigcirc, below=0.4cm of PA2]          (PA3) {Policy\\Analysis 3};
\fi
\node[bigcirc, right=2.4cm of PA1]          (PM1) {\maker\\Maker 1};
\node[bigcirc, right=2.4cm of PA3]          (PM2) {\maker\\Maker 2};
\node[rrect,   right=1.5cm of PM1]          (S)   {Support};
\node[rrect,   right=1.5cm of PM2]          (O)   {Oppose};

\ifdefined\fromanalysis\else
\draw[arr]  (T)   -- (R);
\draw[darr] (R)   -- (PA1);
\draw[arr]  (R)   -- (PA2);
\draw[darr] (R)   -- (PA3);
\fi
\draw[arr]  (PA1) -- node[note, above] {only gains} (PM1);
\draw[arr]  (PA3) -- node[note, above] {only losses} (PM2);
\ifdefined\fromanalysis\else
\node[note, right=0.3cm of PA2] {gains \& losses};
\fi
\draw[arr]  (PM1) -- (S);
\draw[arr]  (PM2) -- (O);

% What citizens see: the rival analyses and what each policy maker does.
% Left out of the competing-from-analysis variant.
\ifdefined\fromanalysis\else
\coordinate (bl) at ($(PA1.north west)+(0,0.5)$);
\draw[decorate, decoration={brace, amplitude=8pt}, thick]
  (bl) -- (bl -| S.east) node[midway, above=10pt] {Observed by citizens};
\fi

\end{tikzpicture}
\end{document}
